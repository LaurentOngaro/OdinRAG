package game

Direction :: enum {
	North,
	East,
	South,
	West,
}

reverse_direction :: proc(dir: Direction) -> Direction {
	switch dir {
		case .North: return .South
		case .East: return .West
		case .South: return .North
		case .West: return .East
	}

	return dir
}

vec2i_from_direction :: proc(dir: Direction) -> Vec2i {
	switch dir {
		case .North:
			return {0, -1}
		case .East:
			return {1, 0}
		case .South:
			return {0, 1}
		case .West:
			return {-1, 0}
	}

	return {0, 0}
}

package game

import "core:log"

DrawableRect :: struct {
	rect: Rect,
	color: Color,
}

DrawableRectLines :: distinct DrawableRect

DrawableTexture :: struct {
	texture: Texture,
	pos: Vec2i,
	flip: bool,
}

Drawable :: union #no_nil {
	DrawableRect,
	DrawableRectLines,
	DrawableTexture,
}

DrawablesArray :: [4096]Drawable

@(private="file")
drawables: ^DrawablesArray
@(private="file")
num_drawables: int

drawables_clear :: proc() {
	num_drawables = 0
}

drawables_init :: proc(d: ^DrawablesArray) {
	drawables = d
}

drawables_slice :: proc() -> []Drawable {
	return drawables[:num_drawables]
}

add_drawable :: proc(d: Drawable) {
	if num_drawables == len(drawables) {
		log.error("Out of drawables!")
		return
	}

	drawables[num_drawables] = d
	num_drawables += 1
}

draw_rect :: proc(r: Rect, c: Color) {
	add_drawable(DrawableRect {
		rect = r,
		color = c,
	})
}

draw_rect_lines :: proc(r: Rect, c: Color) {
	add_drawable(DrawableRectLines {
		rect = r,
		color = c,
	})
}

draw_texture :: proc(t: Texture, pos: Vec2i, flip: bool = false) {
	add_drawable(DrawableTexture {
		texture = t,
		pos = pos,
		flip = flip,
	})
}

update :: proc() {
	if rl.IsKeyPressed(.ONE) {
		g_mem.placement_mode = PlacementModeTrackStart{}
	} else if rl.IsKeyPressed(.TWO) {
		g_mem.placement_mode = PlacementModeStation{}
	} else if rl.IsKeyPressed(.THREE) {
		g_mem.placement_mode = PlacementModeTrain{}
	}

	drawables_clear()

	cam := game_camera()

	mp := rl.GetScreenToWorld2D(rl.GetMousePosition(), cam)
	hover_coord := Vec2i {
		int(mp.x/GridSize),
		int(mp.y/GridSize),
	}

	hover_coord_world := vec2_from_vec2i(hover_coord*GridSize)
	hover_idx := tile_index_from_coord(g_mem.grid, hover_coord)

	switch &p in g_mem.placement_mode {
		case PlacementModeTrackStart:
			hover_world_rect := Rect {
				hover_coord_world.x, hover_coord_world.y,
				GridSize, GridSize,
			}

			r_north, _ := split_rect_top(hover_world_rect, hover_world_rect.height / 3, 0)
			_, r_south := split_rect_bottom(hover_world_rect, hover_world_rect.height / 3, 0)
			r_west, _ := split_rect_left(hover_world_rect, hover_world_rect.width / 3, 0)
			_, r_east := split_rect_right(hover_world_rect, hover_world_rect.width / 3, 0)

			in_north := mouse_in_world_rect(r_north, cam)
			in_east := mouse_in_world_rect(r_east, cam)
			in_south := mouse_in_world_rect(r_south, cam)
			in_west := mouse_in_world_rect(r_west, cam)

			outside_rect := (p.from_dir == .North && !in_north) ||
							(p.from_dir == .East && !in_east) ||
							(p.from_dir == .South && !in_south) ||
							(p.from_dir == .West && !in_west)

			if outside_rect {
				if in_north {
					p.from_dir = .North
				}

				if in_east {
					p.from_dir = .East
				}

				if in_south {
					p.from_dir = .South
				}

				if in_west {
					p.from_dir = .West
				}
			}

			switch p.from_dir {
				case .North: draw_rect(r_north, ColorTrain)
				case .East: draw_rect(r_east, ColorTrain)
				case .South: draw_rect(r_south, ColorTrain)
				case .West: draw_rect(r_west, ColorTrain)
			}

			draw_rect_lines(hover_world_rect, ColorTrain)

			if rl.IsMouseButtonPressed(.LEFT) {
				g_mem.placement_mode = PlacementModeTrackChosingDestDirection {
					from_dir = p.from_dir,
					tile = hover_idx,
				}
			}

		case PlacementModeTrackChosingDestDirection:
			tile_coord := tile_coord_from_index(g_mem.grid, p.tile)
			tile_world_coord := vec2_from_vec2i(tile_coord*GridSize)

			tile_rect := Rect {
				tile_world_coord.x, tile_world_coord.y,
				GridSize, GridSize,
			}

			draw_rect_lines(tile_rect, ColorTrain)

			r_north, _ := split_rect_top(tile_rect, tile_rect.height / 3, 0)
			_, r_south := split_rect_bottom(tile_rect, tile_rect.height / 3, 0)
			r_west, _ := split_rect_left(tile_rect, tile_rect.width / 3, 0)
			_, r_east := split_rect_right(tile_rect, tile_rect.width / 3, 0)

			in_north := mouse_in_world_rect(r_north, cam)
			in_east := mouse_in_world_rect(r_east, cam)
			in_south := mouse_in_world_rect(r_south, cam)
			in_west := mouse_in_world_rect(r_west, cam)

			outside_rect := (p.to_dir == .North && !in_north) ||
							(p.to_dir == .East && !in_east) ||
							(p.to_dir == .South && !in_south) ||
							(p.to_dir == .West && !in_west)

			if outside_rect {
				if in_north && p.from_dir != .North {
					p.to_dir = .North
				}

				if in_east && p.from_dir != .East {
					p.to_dir = .East
				}

				if in_south && p.from_dir != .South {
					p.to_dir = .South
				}

				if in_west && p.from_dir != .West {
					p.to_dir = .West
				}
			}
			
			if p.from_dir != p.to_dir {
				draw_track_tile_directions(tile_coord, p.from_dir, p.to_dir)
			} else {
				switch p.from_dir {
					case .North: draw_rect(r_north, ColorTrain)
					case .East: draw_rect(r_east, ColorTrain)
					case .South: draw_rect(r_south, ColorTrain)
					case .West: draw_rect(r_west, ColorTrain)
				}
			}

			if hover_idx != p.tile && p.from_dir != p.to_dir {
				coord_diff := hover_coord - tile_coord

				if coord_diff.x > 0 {
					p.to_dir = .East
				}
				if coord_diff.x < 0 {
					p.to_dir = .West
				}
				if coord_diff.y > 0 {
					p.to_dir = .South
				}
				if coord_diff.y < 0 {
					p.to_dir = .North
				}

				dir := track_direction_from_directions(p.from_dir, p.to_dir)
				rev_dir := reverse_track_direction(dir)

				switch &v in g_mem.grid.tiles[p.tile] {
					case TileTypeNone:
						g_mem.grid.tiles[p.tile] = TileTypeTrack { directions = { dir, rev_dir}}

					case TileTypeTrack:
						v.directions += {dir, rev_dir}
				}

				g_mem.placement_mode = PlacementModeTrackChosingDestDirection {
					tile = hover_idx,
					from_dir = reverse_direction(p.to_dir),
					to_dir = p.to_dir,
				}

				break
			}

			if rl.IsMouseButtonReleased(.LEFT) {
				if hover_idx == p.tile && p.from_dir != p.to_dir {
					dir := track_direction_from_directions(p.from_dir, p.to_dir)
					rev_dir := reverse_track_direction(dir)

					switch &v in g_mem.grid.tiles[p.tile] {
						case TileTypeNone:
							g_mem.grid.tiles[p.tile] = TileTypeTrack { directions = { dir, rev_dir}}

						case TileTypeTrack:
							v.directions += {dir, rev_dir}
					}
				}

				g_mem.placement_mode = PlacementModeTrackStart {
					from_dir = p.from_dir,
				}
			}

		case PlacementModeStation:
			switch &v in g_mem.grid.tiles[hover_idx] {
				case TileTypeNone:

				case TileTypeTrack:
					if !v.station {
						draw_track_station(hover_coord, v.directions)
					}

					if rl.IsMouseButtonPressed(.LEFT) {
						v.station = !v.station
					}
		}

		case PlacementModeTrain:
			switch &v in g_mem.grid.tiles[hover_idx] {
				case TileTypeNone:

				case TileTypeTrack:
					if rl.IsMouseButtonPressed(.LEFT) {
						if heading := first_track_direction_in_bitset(v.directions); heading != .Invalid {
							t := Train {
								cur_cell = hover_idx,
								heading = heading,
								move_timer = TrainMoveSpeed,
							}
							ha_add(&g_mem.trains, t)
						}
					}

					draw_train(hover_coord)
			}
	}

	if rl.IsMouseButtonDown(.RIGHT) {
		switch p in g_mem.placement_mode {
			case PlacementModeTrain:
				trains_iter := ha_make_iter(&g_mem.trains)
				for t, th in ha_iter_ptr(&trains_iter) {
					if t.cur_cell == hover_idx {
						ha_remove(&g_mem.trains, th)
						break
					}
				}
			case PlacementModeTrackStart:
				g_mem.grid.tiles[hover_idx] = {}
			case PlacementModeTrackChosingDestDirection:
			case PlacementModeStation:
		}
	}

	trains_iter := ha_make_iter(&g_mem.trains)
	for t in ha_iter_ptr(&trains_iter) {
		t.move_timer -= rl.GetFrameTime()
		next_tile, next_heading, next_tile_ok := train_get_next_tile(t^)

		if t.move_timer <= 0 {
			t.move_timer = TrainMoveSpeed + t.move_timer

			if next_tile_ok {
				t.cur_cell = next_tile
				t.heading = next_heading
			} else {
				t.heading = reverse_track_direction(t.heading)
			}
		}
	}
}

draw :: proc() {
	rl.BeginDrawing()
	rl.ClearBackground(ColorGround)
	rl.BeginMode2D(game_camera())

	for c, idx in g_mem.grid.tiles {
		switch v in c {
			case TileTypeNone:
			case TileTypeTrack:
				c := tile_coord_from_index(g_mem.grid, idx)
				draw_track_tile(c, v.directions)

				if v.station {
					draw_track_station(c, v.directions)
				}
		}
	}

	trains_iter := ha_make_iter(&g_mem.trains)
	for t in ha_iter_ptr(&trains_iter) {
		next_tile, _, next_tile_ok := train_get_next_tile(t^)

		cp := tile_coord_from_index(g_mem.grid, t.cur_cell)

		// interpolation
		if next_tile_ok {
			np := tile_coord_from_index(g_mem.grid, next_tile)

			p1 := rl.Vector2 {
				f32(cp.x)*GridSize, f32(cp.y)*GridSize,
			}

			p2 := rl.Vector2 {
				f32(np.x)*GridSize, f32(np.y)*GridSize,
			}

			pos_t := remap(TrainMoveSpeed - t.move_timer, 0, TrainMoveSpeed, 0, 1)
			pos := math.lerp(p1, p2, pos_t)
			draw_train_world_coords(pos)
		} else {
			draw_train(cp)
		}
	}

	drawables := drawables_slice()

	for d in drawables {
		switch v in d {
			case DrawableRect: rl.DrawRectangleRec(v.rect, v.color)
			case DrawableRectLines: rl.DrawRectangleLinesEx(v.rect, 1, v.color)
			case DrawableTexture:
				p := vec2_from_vec2i(v.pos*GridSize)
				dest := Rect {
					p.x, p.y,
					GridSize, GridSize,
				}
				rl.DrawTexturePro(v.texture, texture_rect(v.texture, v.flip), dest, 0, 0, rl.WHITE)
		}
	}

	rl.EndMode2D()
	rl.BeginMode2D(ui_camera())

	#partial switch &p in g_mem.placement_mode {
		case PlacementModeTrackStart:
			rl.DrawTextEx(rl.GetFontDefault(), "Track", {2, 2}, 8, 1, rl.BLACK)
			rl.DrawTextEx(rl.GetFontDefault(), fmt.ctprintf("%v", p.from_dir), {100, 2}, 8, 1, rl.BLACK)

		case PlacementModeTrackChosingDestDirection:
			rl.DrawTextEx(rl.GetFontDefault(), "Track", {2, 2}, 8, 1, rl.BLACK)
			rl.DrawTextEx(rl.GetFontDefault(), fmt.ctprintf("%v", p.from_dir), {100, 2}, 8, 1, rl.BLACK)
			rl.DrawTextEx(rl.GetFontDefault(), fmt.ctprintf("%v", p.to_dir), {140, 2}, 8, 1, rl.BLACK)

		case:
			placement_mode_str := strings.trim_prefix(fmt.tprint(union_type(g_mem.placement_mode)), "PlacementMode")
			rl.DrawTextEx(rl.GetFontDefault(), temp_cstring(placement_mode_str), {2, 2}, 8, 1, rl.BLACK)
	}

	rl.EndMode2D()
	rl.EndDrawing()
}

package game

Handle :: struct($T: typeid) {
	// idx 0 means unused. Note that slot 0 is a dummy slot, it can never be used.
	idx: u32,
	gen: u32,
}

HandleArrayItem :: struct($T: typeid) {
	item: T,
	handle: Handle(T),
}

HandleArray :: struct($T: typeid, $N: int) {
	items: #soa[N]HandleArrayItem(T),
	num_items: u32,
}

ha_add :: proc(a: ^HandleArray($T, $N), v: T) -> (Handle(T), bool) #optional_ok {
	for idx in 1..<a.num_items {
		i := &a.items[idx]

		if idx != 0 && i.handle.idx == 0 {
			i.handle.idx = u32(idx)
			i.item = v
			return i.handle, true
		}
	}

	// Index 0 is dummy
	if a.num_items == 0 {
		a.num_items += 1
	}

	if a.num_items == len(a.items) {
		return {}, false
	}

	idx := a.num_items
	i := &a.items[a.num_items]
	a.num_items += 1
	i.handle.idx = idx
	i.handle.gen = 1
	i.item = v
	return i.handle, true
}

ha_get :: proc(a: HandleArray($T, $N), h: Handle(T)) -> (T, bool) {
	if h.idx == 0 {
		return {}, false
	}

	if int(h.idx) < len(a.items) && int(h.idx) < a.num_items && a.items[h.idx].handle == h {
		return ha.items[h.idx], true
	}

	return {}, false
}

ha_get_ptr :: proc(a: HandleArray($T, $N), h: Handle(T)) -> ^T {
	if h.idx == 0 {
		return nil
	}

	if int(h.idx) < len(a.items) && int(h.idx) < a.num_items && a.items[h.idx].handle == h {
		return &ha.items[h.idx].item
	}

	return nil
}

ha_remove :: proc(a: ^HandleArray($T, $N), h: Handle(T)) {
	if h.idx == 0 {
		return
	}

	if int(h.idx) < len(a.items) && h.idx < a.num_items && a.items[h.idx].handle == h {
		a.items[h.idx].handle.idx = 0
		a.items[h.idx].handle.gen += 1
	}
}

ha_valid :: proc(a: HandleArray($T, $N), h: Handle(T)) -> bool {
	if h.idx == 0 {
		return false
	}

	return int(h.idx) < len(a.items) && int(h.idx) < a.num_items && a.items[h.idx].handle == h
}

HandleArrayIter :: struct($T: typeid, $N: int) {
	a: ^HandleArray(T, N),
	index: int,
}

ha_make_iter :: proc(a: ^HandleArray($T, $N)) -> HandleArrayIter(T, N) {
	return HandleArrayIter(T, N) { a = a }
}

ha_iter :: proc(it: ^HandleArrayIter($T, $N)) -> (val: T, h: Handle(T), cond: bool) {
	cond = it.index < it.a.num_items

	for ; cond; cond = it.index < it.a.num_items {
		if it.ha.items[it.index].handle.idx == 0 {
			it.index += 1
			continue
		}

		val = it.a.items[it.index].item
		h = it.a.items[it.index].handle
		it.index += 1
		break
	}

	return
}

ha_iter_ptr :: proc(it: ^HandleArrayIter($T, $N)) -> (val: ^T, h: Handle(T), cond: bool) {
	cond = it.index < int(it.a.num_items)

	for ; cond; cond = it.index < int(it.a.num_items) {
		if it.a.items[it.index].handle.idx == 0 {
			it.index += 1
			continue
		}

		val = &it.a.items[it.index].item
		h = it.a.items[it.index].handle
		it.index += 1
		break
	}

	return
}

// generic odin helpers

package game

import "core:reflect"
import "core:strings"
import "core:intrinsics"

increase_or_wrap_enum :: proc(e: $T) -> T {
	ei := int(e) + 1

	if ei >= len(T) {
		ei = 0
	}

	return T(ei)
}

union_type :: proc(a: any) -> typeid {
	return reflect.union_variant_typeid(a)
}

temp_cstring :: proc(s: string) -> cstring {
	return strings.clone_to_cstring(s, context.temp_allocator)
}

remap :: proc "contextless" (old_value, old_min, old_max, new_min, new_max: $T) -> (x: T) where intrinsics.type_is_numeric(T), !intrinsics.type_is_array(T) {
	old_range := old_max - old_min
	new_range := new_max - new_min
	if old_range == 0 {
		return new_range / 2
	}

	init: proc(),

	shutdown_window: proc(),

load_game_api :: proc(api_version: int) -> (api: GameAPI, ok: bool) {
	mod_time, mod_time_error := os.last_write_time_by_name("game.dll")

	if mod_time_error != os.ERROR_NONE {
		return
	}

	game_dll_name := fmt.tprintf("game_{0}.dll", api_version)

	if libc.system(fmt.ctprintf("copy game.dll {0}", game_dll_name)) != 0 {
		fmt.println("Failed to copy game.dll to {0}", game_dll_name)
		return
	}

	_, ok = dynlib.initialize_symbols(&api, game_dll_name, "game_", "lib")
	api.api_version = api_version
	api.modification_time = mod_time
	ok = true

	return
}

unload_game_api :: proc(api: ^GameAPI) {
	if api.lib != nil {
		dynlib.unload_library(api.lib)
	}

	if libc.system(fmt.ctprintf("del game_{0}.dll", api.api_version)) != 0 {
		fmt.println("Failed to remove game_{0}.dll copy", api.api_version)
	}
}

		err := false

			fmt.printf("%v: Leaked %v bytes\n", value.location, value.size)
			err = true
		}

		tracking_allocator_clear(a)
		return err
	}

		fmt.println("Failed to load Game API")
		return
	}

	game_api.init()

	old_game_apis := make([dynamic]GameAPI, default_allocator)

	ft_avg: [60]f32
	ft_avg_i: int
	last_spike: f32

	window_open := true
	for window_open {
		window_open = game_api.update()
		force_reload := rl.IsKeyPressed(.F5)
		force_restart := rl.IsKeyPressed(.F6)
		reload := force_reload || force_restart

		if game_dll_mod_err == os.ERROR_NONE && game_api.modification_time != game_dll_mod {

		}

					for &g in old_game_apis {

package main

import "core:runtime"
import "core:mem"
import "core:sync"
import "core:fmt"

Tracking_Allocator_Entry :: struct {
	memory:    rawptr,
	size:      int,
	alignment: int,
	mode:      mem.Allocator_Mode,
	err:       mem.Allocator_Error,
	location:  runtime.Source_Code_Location,
}

Tracking_Allocator :: struct {
	backing:           mem.Allocator,
	allocation_map:    map[rawptr]Tracking_Allocator_Entry,
	mutex:             sync.Mutex,
	clear_on_free_all: bool,
}

tracking_allocator_init :: proc(t: ^Tracking_Allocator, backing_allocator: mem.Allocator, internals_allocator := context.allocator) {
	t.backing = backing_allocator
	t.allocation_map.allocator = internals_allocator

	if .Free_All in mem.query_features(t.backing) {
		t.clear_on_free_all = true
	}
}

tracking_allocator_destroy :: proc(t: ^Tracking_Allocator) {
	delete(t.allocation_map)
}


tracking_allocator_clear :: proc(t: ^Tracking_Allocator) {
	sync.mutex_lock(&t.mutex)
	clear(&t.allocation_map)
	sync.mutex_unlock(&t.mutex)
}


@(require_results)
allocator_from_tracking_allocator :: proc(data: ^Tracking_Allocator) -> mem.Allocator {
	return mem.Allocator{
		data = data,
		procedure = tracking_allocator_proc,
	}
}

tracking_allocator_proc :: proc(allocator_data: rawptr, mode: mem.Allocator_Mode,
                                size, alignment: int,
                                old_memory: rawptr, old_size: int, loc := #caller_location) -> (result: []byte, err: mem.Allocator_Error) {
	data := (^Tracking_Allocator)(allocator_data)

	sync.mutex_guard(&data.mutex)

	if mode == .Query_Info {
		info := (^mem.Allocator_Query_Info)(old_memory)
		if info != nil && info.pointer != nil {
			if entry, ok := data.allocation_map[info.pointer]; ok {
				info.size = entry.size
				info.alignment = entry.alignment
			}
			info.pointer = nil
		}

		return
	}

	if mode == .Free && old_memory != nil && old_memory not_in data.allocation_map {
		fmt.panicf("At %v: Bad free!", loc)
	} else {
		result = data.backing.procedure(data.backing.data, mode, size, alignment, old_memory, old_size, loc) or_return
	}
	result_ptr := raw_data(result)

	if data.allocation_map.allocator.procedure == nil {
		data.allocation_map.allocator = context.allocator
	}

	switch mode {
	case .Alloc, .Alloc_Non_Zeroed:
		data.allocation_map[result_ptr] = Tracking_Allocator_Entry{
			memory = result_ptr,
			size = size,
			mode = mode,
			alignment = alignment,
			err = err,
			location = loc,
		}
	case .Free:
		delete_key(&data.allocation_map, old_memory)
	case .Free_All:
		if data.clear_on_free_all {
			clear_map(&data.allocation_map)
		}	
	case .Resize, .Resize_Non_Zeroed:
		if old_memory != result_ptr {
			delete_key(&data.allocation_map, old_memory)
		}
		data.allocation_map[result_ptr] = Tracking_Allocator_Entry{
			memory = result_ptr,
			size = size,
			mode = mode,
			alignment = alignment,
			err = err,
			location = loc,
		}

	case .Query_Features:
		set := (^mem.Allocator_Mode_Set)(old_memory)
		if set != nil {
			set^ = {.Alloc, .Alloc_Non_Zeroed, .Free, .Free_All, .Resize, .Query_Features, .Query_Info}
		}
		return nil, nil

	case .Query_Info:
		unreachable()
	}

	return
}
